/** * Guarantee Tracker <-> Google Sheet Two-way Sync (v2 - แก้ปัญหา CORS) * วางโค้ดนี้ทับของเดิมใน Extensions > Apps Script ของ Google Sheet */ const SHEET_NAME = 'ชีต1'; // ถ้าชื่อแท็บชีตไม่ใช่ "ชีต1" ให้แก้ตรงนี้ function getSheet_() { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); } // ── อ่านข้อมูลทั้งหมด (รองรับ JSONP เพื่อเลี่ยงปัญหา CORS) ────── function doGet(e) { const sheet = getSheet_(); const data = sheet.getDataRange().getValues(); const rows = []; for (let i = 1; i < data.length; i++) { const r = data[i]; if (!r[0]) continue; // ข้ามแถวว่าง rows.push({ row: i + 1, lg: String(r[0] || ''), name: String(r[1] || ''), amount: r[2] || 0, startDate: formatDate_(r[3]), expiry: formatDate_(r[4]), status: r[5] || 'active', bank: String(r[6] || '').trim().toUpperCase() }); } const json = JSON.stringify({ rows: rows }); const callback = e.parameter && e.parameter.callback; if (callback) { return ContentService.createTextOutput(callback + '(' + json + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return jsonOut_({ rows: rows }); } // ── รับข้อมูลจากเว็บมาอัปเดต/เพิ่ม/ลบแถวในชีต (จับคู่ด้วยเลข LG) ── function doPost(e) { const sheet = getSheet_(); const payload = JSON.parse(e.postData.contents); const action = payload.action; if (action === 'upsert') { const item = payload.item; const targetRow = findRowByLg_(sheet, item.lg) || (sheet.getLastRow() + 1); sheet.getRange(targetRow, 1, 1, 7).setValues([[ item.lg || '', item.name || '', item.amount || '', item.startDate || '', item.expiry || '', item.status || 'active', item.bank || '' ]]); return jsonOut_({ success: true, row: targetRow }); } if (action === 'delete') { const row = payload.row || findRowByLg_(sheet, payload.lg); if (row) sheet.deleteRow(row); return jsonOut_({ success: true }); } return jsonOut_({ success: false, error: 'unknown action' }); } function findRowByLg_(sheet, lg) { if (!lg) return null; const lastRow = sheet.getLastRow(); if (lastRow < 2) return null; const data = sheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (let i = 0; i < data.length; i++) { if (String(data[i][0]).trim() === String(lg).trim()) return i + 2; } return null; } function formatDate_(d) { if (Object.prototype.toString.call(d) === '[object Date]') { return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } return d || ''; } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
